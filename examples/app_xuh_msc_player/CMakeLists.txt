option(APP_XUH_MSC_PLAYER_AUDIO_METER "Show the stereo audio level meter in the top-right of the LCD (mpeg1_audio)" ON)
option(APP_XUH_MSC_PLAYER_DECODE_LEAD "Show estimated audio decode lead over displayed video (mpeg1_audio)" ON)
if(NOT APP_XUH_MSC_PLAYER_DECODE_LEAD)
    list(APPEND APP_COMPILER_FLAGS -DAPP_XUH_MSC_PLAYER_DECODE_LEAD=0)
endif()
if(NOT APP_XUH_MSC_PLAYER_AUDIO_METER)
    list(APPEND APP_COMPILER_FLAGS -DAPP_XUH_MSC_PLAYER_AUDIO_METER=0)

if(APP_XUH_MSC_PLAYER_FPS_OVERLAY)
    list(APPEND APP_COMPILER_FLAGS -DAPP_XUH_MSC_PLAYER_FPS_OVERLAY=1)

    list(APPEND APP_COMPILER_FLAGS -DAPP_XUH_MSC_PLAYER_FPS_OVERLAY=0)

option(APP_XUH_MSC_PLAYER_AUDIO_DECODE "Decode MP2 in mpeg1_audio (OFF measures MPG demux/I/O only)" ON)

    list(APPEND APP_COMPILER_FLAGS -DAPP_XUH_MSC_PLAYER_PACING_STRICT=1)

if(NOT APP_XUH_MSC_PLAYER_MPEG_FILE STREQUAL "")

        message(FATAL_ERROR "APP_XUH_MSC_PLAYER_MPEG_FILE must be a filename without a path")

    list(APPEND APP_DEPENDENT_MODULES "${XUH_MPEG_DEPENDENCY}")
    set(APP_COMPILER_FLAGS_mpeg1 ${APP_COMPILER_FLAGS})
    list(FILTER APP_COMPILER_FLAGS_mpeg1 EXCLUDE REGEX "^-DAPP_XUH_MSC_PLAYER_ENABLE_JPEG=")
    list(APPEND APP_COMPILER_FLAGS_mpeg1 -DAPP_XUH_MSC_PLAYER_ENABLE_JPEG=0 -DAPP_XUH_MSC_PLAYER_ENABLE_MPEG1=1)
    set(APP_COMPILER_FLAGS_mpeg1_audio ${APP_COMPILER_FLAGS_mpeg1} -DAPP_XUH_MSC_PLAYER_MPEG_AUDIO=1)

        list(APPEND APP_COMPILER_FLAGS_mpeg1_audio -DAPP_XUH_MSC_PLAYER_AUDIO_DECODE=0)
    endif()
endif()

XMOS_REGISTER_APP()
# XTC 15.3.1 resolves modules for all configurations together. Keep each
# decoder's sources out of the other executable, not merely linker-discarded.
foreach(target ${APP_BUILD_TARGETS})
    get_target_property(sources ${target} SOURCES)
    if(target STREQUAL "app_xuh_msc_player_jpeg")
        list(FILTER sources EXCLUDE REGEX "/lib_mpeg/lib_mpeg/")
        set(configuration jpeg)
    else()
        list(FILTER sources EXCLUDE REGEX "/lib_jpeg/lib_jpeg/")
        if(target STREQUAL "app_xuh_msc_player_mpeg1_audio")
            set(configuration mpeg1_audio)
        else()
            set(configuration mpeg1)
        endif()
    endif()
    if(NOT target STREQUAL "app_xuh_msc_player_mpeg1_audio")
        list(FILTER sources EXCLUDE REGEX "/(audio_pipeline|mpeg_ps_reader)\\.c$")
    endif()
    if(configuration STREQUAL "jpeg")
        list(FILTER sources EXCLUDE REGEX "/pl_mpeg_idct_xs3\\.S$")
    endif()
    set_property(TARGET ${target} PROPERTY SOURCES "${sources}")
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${configuration}")
endforeach()
