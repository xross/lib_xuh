cmake_minimum_required(VERSION 3.21)
include($ENV{XMOS_CMAKE_PATH}/xcommon.cmake)
project(test_activity_led)
set(XMOS_SANDBOX_DIR ${CMAKE_CURRENT_LIST_DIR}/../../../../../)
set(APP_HW_TARGET XK-EVK-XU316 CACHE STRING "XU316 target; the test drives tile-1 port 32A as on the AIV board")
set(APP_INCLUDES src ../../src)
set(APP_C_SRCS src/test.c ../../src/activity_led.c)
set(APP_XC_SRCS src/main.xc)
set(APP_COMPILER_FLAGS -O3 -g -report -DAPP_XUH_MSC_PLAYER_ACTIVITY_LED=1)
option(APP_XUH_MSC_PLAYER_LED_ACTIVE_LOW "Invert both activity LED outputs" OFF)
if(APP_XUH_MSC_PLAYER_LED_ACTIVE_LOW)
    list(APPEND APP_COMPILER_FLAGS -DAPP_XUH_MSC_PLAYER_LED_ACTIVE_LOW=1)
endif()
set(APP_DEPENDENT_MODULES "")
XMOS_REGISTER_APP()
set_target_properties(${APP_BUILD_TARGETS} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
