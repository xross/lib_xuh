cmake_minimum_required(VERSION 3.21)
project(app_xuh_player_native_tests C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(FFMPEG_EXECUTABLE ffmpeg REQUIRED)
set(FATFS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../lib_fatfs" CACHE PATH "lib_fatfs repository for the MJPEG reader's FatFs types")
if(NOT EXISTS "${FATFS_ROOT}/lib_fatfs/api/ff.h")
    message(FATAL_ERROR "Set FATFS_ROOT to a lib_fatfs checkout with its FatFs submodule initialised")
endif()
option(APP_TEST_SANITIZE "Enable native address/undefined sanitizers" ON)
add_executable(mpeg_ps_reader_test mpeg_ps_reader_test.c ../src/mpeg_ps_reader.c)
add_executable(mpeg_ps_demux mpeg_ps_demux.c ../src/mpeg_ps_reader.c)
add_executable(fps_overlay_test fps_overlay_test.c ../src/fps_overlay.c)
add_executable(video_window_test video_window_test.c ../src/fps_overlay.c)
add_executable(frame_timing_test frame_timing_test.c)
add_executable(activity_led_test activity_led_test.c ../src/activity_led.c)
add_executable(mjpeg_reader_test mjpeg_reader_test.c ../src/mjpeg_reader.c)
target_include_directories(mjpeg_reader_test PRIVATE "${FATFS_ROOT}/lib_fatfs/api")
target_compile_definitions(fps_overlay_test PRIVATE APP_XUH_PLAYER_ENABLE_JPEG=0
    APP_XUH_PLAYER_ENABLE_MPEG1=1)
target_compile_definitions(video_window_test PRIVATE APP_XUH_PLAYER_ENABLE_JPEG=0
    APP_XUH_PLAYER_ENABLE_MPEG1=1)
foreach(target mpeg_ps_reader_test mpeg_ps_demux fps_overlay_test video_window_test frame_timing_test activity_led_test mjpeg_reader_test)
    target_include_directories(${target} PRIVATE ../src)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    if(APP_TEST_SANITIZE)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
    if(NOT target STREQUAL "mpeg_ps_demux")
        add_test(NAME ${target} COMMAND ${target})
        set_tests_properties(${target} PROPERTIES TIMEOUT 30 ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1")
    endif()
endforeach()
add_test(NAME mpeg_ps_ffmpeg COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_LIST_DIR}/test_ps_demux.py --demux $<TARGET_FILE:mpeg_ps_demux>
    --ffmpeg "${FFMPEG_EXECUTABLE}" --work ${CMAKE_CURRENT_BINARY_DIR}/ps-fixtures)
set_tests_properties(mpeg_ps_ffmpeg PROPERTIES TIMEOUT 120 ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1")
add_test(NAME profile_summary COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_profile_summary.py)
set_tests_properties(profile_summary PROPERTIES TIMEOUT 30)
